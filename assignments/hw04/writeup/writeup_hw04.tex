\section*{Problem 1}
TISP variable selection method ith threshold operator:
\[
\Theta(x,\lambda,\eta) = 
\begin{cases} 
    0 & \text{if} |x|\leq \lambda \\
    \frac{x}{1+\eta} & |x| > \lambda
\end{cases}
\]
Penalized logistic regression:
\[
L(\boldsymbol{\omega}) = \frac{1}{N}\sum_{j=1}^{N}\text{ln}
(1+e^{-y_j^{*}\boldsymbol{\omega}^T\boldsymbol{x}_j}) + \lambda\sum_{i=1}^{p}p(\omega_i)
\]
assuming \(y_j^*\in \{-1,1\}\)
and algorithm iteration:
\[
\boldsymbol{\omega}^{t+1} = \Theta\left(\boldsymbol{\omega}^{t} + 
\eta'X^T\left[\boldsymbol{y}- \frac{1}{1 + e^{-X\boldsymbol{\omega}^{t}}}\right], \lambda\right)
\]
assuming \(y_j\in \{-0,1\}\).

\section*{Problem 2}
FSA variable selection using logistic loss:
\[
L_D(\beta) = \frac{1}{N}\sum_{i=1}^{N}\text{ln}
(1+e^{-y_i\beta^T\boldsymbol{x}_j}) + s\lvert\lvert\beta\rvert\rvert_{2}^2
\]
We can compute the gradient for one element with a few key facts, and a substitution: \(u = -y_i\beta^T\boldsymbol{x}_j\):

\[
\frac{d}{du}\text{log}(1 + e^{u}) = \frac{e^{u}}{1 + e^{u}} = \frac{1}{1 + e^{-u}}\quad \quad\quad
\frac{d}{d\beta}u = -y_i\boldsymbol{x}_i
\]
\[
\frac{d}{d\beta}\text{log}(1 + e^{u}) = \frac{1}{1 + e^{-u}}(-y_i\boldsymbol{x}_i)
\]
substitution gives us:
\[
\frac{d}{d\beta}\text{log}(1 + e^{-y_i\beta^T\boldsymbol{x}_j}) = 
\frac{1}{1 + e^{y_i\beta^T\boldsymbol{x}_j}}(-y_i\boldsymbol{x}_i) + 2s\beta
\]
With this, we can get the gradient for all N:
\[
\nabla L_D(\beta) = \frac{-1}{N}\sum_{i=1}^{N}\frac{(y_i\boldsymbol{x}_i)}{1 + e^{y_i\beta^T\boldsymbol{x}_j}}
\]
We also use an inverse scheduler with parameter \(\mu\):
\[
M_i = k + (p-k)\text{max}\left(0, \frac{N^{iter}-2i}{2i\mu+N^{iter}}\right)
\]
